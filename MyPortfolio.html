    events.sk
    

    on break:
    set {_blockName} to "%type of event-block%"

    if {_blockName} contains "glazed terracotta" or "emerald block" or "diamond block" or "gold block" or "coal block" or "iron block" or "redstone block" or "lapis block":

        set {_type} to type of event-block

        cancel event
        set block at event-location to air

        set {_fortune} to level of fortune on player's tool
        if {_fortune} is not set:
            set {_fortune} to 0

        set {_maxDrop} to 1 + {_fortune}
        set {_amount} to a random integer between 1 and {_maxDrop}

        give {_amount} of {_type} to player

on death of player:
    set {_drops::*} to drops
    clear drops

    set {_loc} to location of victim
    loop {_drops::*}:
        drop loop-value at {_loc}

on place of cobweb:
    wait 20 seconds
    if block at event-location is cobweb:
        set block at event-location to air

on place:
    if player's gamemode is survival:
        if event-block is not cobweb:
            cancel event

function val(q) {
  if (q.type === "yesno") {
    var c = form.querySelector('input[name="proof"]:checked');
    return c ? c.value : "";
  }
  return document.getElementById("f-" + q.id).value.trim();
}

function updateProgress() {
  var n = 0;
  Q.forEach(function (q, i) {
    var on = val(q) !== "";
    if (on) n++;
    blocksEl.children[i].className = on ? "on" : "";
  });
  document.getElementById("count").textContent = T[lang].progress(n);
}

function applyLang(l) {
  lang = l;
  var t = T[l];
  var root = document.documentElement;
  root.lang = l; root.dir = l === "ar" ? "rtl" : "ltr";
  document.getElementById("btn-ar").setAttribute("aria-pressed", String(l === "ar"));
  document.getElementById("btn-en").setAttribute("aria-pressed", String(l === "en"));
  document.getElementById("brand").textContent = t.brand;
  document.getElementById("title").textContent = t.title;
  document.getElementById("intro").textContent = t.intro;
  var qs = form.querySelectorAll(".qt");
  for (var i = 0; i < qs.length; i++) qs[i].textContent = t.q[i];
  Q.forEach(function (q, i) {
    var el = document.getElementById("f-" + q.id);
    if (el && t.ph[i] !== undefined) el.placeholder = t.ph[i];
  });
  form.querySelector(".t-yes").textContent = t.yes;
  form.querySelector(".t-no").textContent = t.no;
  document.getElementById("submit").textContent = t.submit;
  document.getElementById("res-title").textContent = t.resTitle;
  document.getElementById("res-note").textContent = t.resNote;
  Q.forEach(function (q) { var e = document.getElementById("e-" + q.id); if (!e.hidden) showErr(q, e.dataset.kind); });
  if (!document.getElementById("result").hidden) buildResult();
  updateProgress();
  store("staff-app-lang", l);
}

function showErr(q, kind) {
  var e = document.getElementById("e-" + q.id);
  e.dataset.kind = kind;
  e.textContent = kind === "age" ? T[lang].ageBad : T[lang].required;
  e.hidden = false;
  var f = document.getElementById("f-" + q.id);
  if (f) f.setAttribute("aria-invalid", "true");
}
function clearErr(q) {
  var e = document.getElementById("e-" + q.id);
  e.hidden = true;
  var f = document.getElementById("f-" + q.id);
  if (f) f.removeAttribute("aria-invalid");
}

function saveDraft() {
  var o = {};
  Q.forEach(function (q) { o[q.id] = val(q); });
  store("staff-app-draft", JSON.stringify(o));
}

form.addEventListener("input", function () {
  Q.forEach(function (q) { if (val(q) !== "") clearErr(q); });
  updateProgress(); saveDraft();
});
form.addEventListener("change", function () { updateProgress(); saveDraft(); });

function buildResult() {
  var t = T[lang];
  var lines = [t.docTitle, ""];
  Q.forEach(function (q, i) {
    var v = val(q);
    if (q.type === "yesno") v = t.yn[v] || "";
    lines.push((i + 1) + ". " + t.q[i]);
    lines.push(v);
    lines.push("");
  });
  document.getElementById("res-text").value = lines.join("\n").replace(/\s+$/, "");
}

form.addEventListener("submit", function (ev) {
  ev.preventDefault();
  var first = null;
  Q.forEach(function (q) {
    var v = val(q), bad = null;
    if (v === "") bad = "required";
    else if (q.type === "number") {
      var n = Number(v);
      if (!(n >= 8 && n <= 99)) bad = "age";
    }
    if (bad) { showErr(q, bad); if (!first) first = q; } else clearErr(q);
  });
  if (first) {
    var target = first.type === "yesno" ? document.getElementById("proof-yes") : document.getElementById("f-" + first.id);
    target.focus();
    return;
  }
  buildResult();
  form.hidden = true;
  var r = document.getElementById("result");
  r.hidden = false;
  r.scrollIntoView({ block: "start" });
});
